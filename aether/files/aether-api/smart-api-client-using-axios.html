import axios from "axios";
import type { AxiosError, AxiosInstance } from "axios";

const BASE_URL = "https://p5wrsahdo1.execute-api.us-east-2.amazonaws.com/api/v1";

export interface UserToken {
    accessToken: string;
    refreshToken: string;
    expirationDate: string;
}

class TokenStorage {

    private token?: UserToken;

    getToken() {
        return this.token;
    }

    setToken(token: UserToken) {
        this.token = token;
    }

    clear() {
        this.token = undefined;
    }

}

export const tokenStorage = new TokenStorage();

class ApiService {

    private api: AxiosInstance;

    private isRefreshing = false;

    private refreshPromise: Promise<void> | null = null;

    constructor() {

        this.api = axios.create({
            baseURL: BASE_URL,
            timeout: 30000,
        });

        this.initializeInterceptors();
    }

    get client() {
        return this.api;
    }

    private initializeInterceptors() {

        this.api.interceptors.request.use(async (config) => {

            const token = tokenStorage.getToken();

            if (token) {
                config.headers.Authorization = `Bearer ${token.accessToken}`;
            }

            return config;
        });

        this.api.interceptors.response.use(
            (response) => response,
            this.handleError.bind(this)
        );
    }

    private async handleError(error: AxiosError) {

        const originalRequest: any = error.config;

        if (!originalRequest) {
            return Promise.reject(error);
        }

        if (
            (error.response?.status === 401 ||
                error.response?.status === 403) &&
            !originalRequest._retry
        ) {

            originalRequest._retry = true;

            try {

                await this.refreshAccessToken();

                const token = tokenStorage.getToken();

                if (token) {
                    originalRequest.headers.Authorization =
                        `Bearer ${token.accessToken}`;
                }

                return this.api(originalRequest);

            } catch (err) {

                tokenStorage.clear();

                return Promise.reject(err);
            }

        }

        return Promise.reject(error);
    }

    private async refreshAccessToken() {

        if (this.isRefreshing && this.refreshPromise) {
            return this.refreshPromise;
        }

        this.isRefreshing = true;

        this.refreshPromise = (async () => {

            const token = tokenStorage.getToken();

            if (!token) {
                throw new Error("No refresh token.");
            }

            const response = await axios.post(
                `${BASE_URL}/users/refresh-token`,
                {},
                {
                    headers: {
                        Authorization: `Bearer ${token.accessToken}`,
                    },
                }
            );

            const newToken = response.data.data;

            tokenStorage.setToken({
                accessToken: newToken.access_token,
                refreshToken: token.refreshToken,
                expirationDate: newToken.expiration_date,
            });

        })();

        try {

            await this.refreshPromise;

        } finally {

            this.isRefreshing = false;
            this.refreshPromise = null;

        }

    }

}

export default new ApiService();

This code is creating a smart API client using Axios. Its job is to:

  1. Automatically attach the user's access token to every request.
  2. Detect when the access token expires.
  3. Automatically get a new access token using the refresh endpoint.
  4. Retry the original request without the user noticing.

Think of it like this:

Your access token is a temporary ID card. When it expires, this code automatically goes to the security office, gets a new ID card, and continues what it was doing.

Step 1. Import Axios


import axios from "axios";
import type { AxiosError, AxiosInstance } from "axios";
  • axios sends HTTP requests.
  • AxiosError is the error type.
  • AxiosInstance is a custom Axios client.

Step 2. API URL


const BASE_URL =
"https://p5wrsahdo1.execute-api.us-east-2.amazonaws.com/api/v1";

Instead of writing


https://p5wrsahdo1.execute-api.us-east-2.amazonaws.com/api/v1/signin

every time, you only write


api.client.post("/signin")

because Axios automatically adds the base URL.

Step 3. Token Interface

Step 6. Interceptors

This is the most important part.

Think of interceptors as security guards.

Every request passes through them.

  |

Request Interceptor


this.api.interceptors.request.use(...)

Runs before every request.

It gets the token.


const token = tokenStorage.getToken();

Suppose


accessToken = abc123

Then it adds


Authorization: Bearer abc123

to every request.

So when you write


api.client.get("/users")

Axios actually sends


GET /users

Authorization: Bearer abc123

automatically.

You don't have to write the Authorization header every time.
