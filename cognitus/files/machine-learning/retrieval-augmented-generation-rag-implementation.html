A complete Retrieval-Augmented Generation (RAG) implementation has five main parts:

  1. Document ingestion
  2. Chunking and embedding generation
  3. Vector database
  4. Retriever
  5. LLM generation

Below is a production-style architecture using Node.js, AWS Lambda, OpenAI embeddings, and MongoDB Atlas Vector Search (fits well with your existing AWS serverless + Node.js stack).

Load PDF


import fs from "fs";
import pdf from "pdf-parse";

export async function loadPDF(file) {

    const data = await pdf(fs.readFileSync(file));

    return data.text;

}

Split into Chunks

Simple recursive splitter


export function splitText(text, chunkSize = 800, overlap = 200) {

    const chunks = [];

    let start = 0;

    while (start < text.length) {

        chunks.push(
            text.substring(start, start + chunkSize)
        );

        start += chunkSize - overlap;
    }

    return chunks;

}

Generate Embeddings


import OpenAI from "openai";

const client = new OpenAI({
    apiKey: process.env.OPENAI_API_KEY
});

export async function createEmbedding(text) {

    const result = await client.embeddings.create({

        model: "text-embedding-3-small",

        input: text

    });

    return result.data[0].embedding;

}

Ingest Documents


import Chunk from "../db/chunk.js";

import {loadPDF} from "./loader.js";

import {splitText} from "./splitter.js";

import {createEmbedding} from "./embedder.js";

export async function ingest(file) {

    const text = await loadPDF(file);

    const chunks = splitText(text);

    let index = 0;

    for (const chunk of chunks) {

        const embedding = await createEmbedding(chunk);

        await Chunk.create({

            documentId: file,

            chunkId: index++,

            text: chunk,

            embedding

        });

    }

}

Retrieve Similar Chunks


import Chunk from "../db/chunk.js";

import {createEmbedding} from "../ingest/embedder.js";

export async function retrieve(question) {

    const embedding = await createEmbedding(question);

    const result = await Chunk.aggregate([

        {
            $vectorSearch: {

                index: "vector_index",

                path: "embedding",

                queryVector: embedding,

                numCandidates: 100,

                limit: 5

            }
        }

    ]);

    return result;

}

Build Prompt


export function buildPrompt(question, docs) {

    const context = docs

        .map(d => d.text)

        .join("\n\n");

    return `

You are a helpful assistant.

Only answer using the supplied context.

If the answer does not exist,

say you don't know.

Context:

${context}

Question:

${question}

`;

}

In short:

  • map(d => d.text) extracts the text field from each document.
  • join("\n\n") combines all those text strings into one large string with blank lines between them.
  • The resulting context string is typically inserted into an LLM prompt so the model can answer using the retrieved documents.

Ask GPT


import OpenAI from "openai";

const client = new OpenAI();

export async function answer(prompt) {

    const response = await client.responses.create({

        model: "gpt-5.5",

        input: prompt

    });

    return response.output_text;

}

Complete Flow


import {retrieve} from "./search/retrieve.js";

import {buildPrompt} from "./llm/prompt.js";

import {answer} from "./llm/answer.js";

export async function ask(question) {

    const docs = await retrieve(question);

    const prompt = buildPrompt(question, docs);

    return await answer(prompt);

}

Improving Retrieval

Basic cosine similarity can be enhanced with:

  • Metadata filtering (e.g., document type, author, date)
  • Hybrid search (keyword + vector search)
  • Query rewriting for ambiguous questions
  • Re-ranking using a cross-encoder model
  • Dynamic top-K retrieval based on similarity scores
  • Context compression to reduce token usage
  • Deduplication of overlapping chunks
  • Parent-child retrieval (retrieve small chunks, expand to larger context)
