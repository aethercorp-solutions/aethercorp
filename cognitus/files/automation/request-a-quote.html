Industry Standard Next.js Form


"use client";

import { useState } from "react";

export default function QuotePage() {
  const [loading, setLoading] = useState(false);

  async function handleSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();

    setLoading(true);

    const formData = new FormData(e.currentTarget);

    const response = await fetch("/api/quote", {
      method: "POST",
      body: formData,
    });

    if (response.ok) {
      alert("Quote request submitted successfully.");
      e.currentTarget.reset();
    } else {
      alert("Failed to submit quote request.");
    }

    setLoading(false);
  }

  return (
    <div className="max-w-3xl mx-auto p-6">
      <h1 className="text-3xl font-bold mb-6">
        Request a Quote
      </h1>

      <form
        onSubmit={handleSubmit}
        className="space-y-4"
      >
        <input
          name="name"
          placeholder="Full Name"
          required
          className="w-full border p-3 rounded"
        />

        <input
          name="company"
          placeholder="Company Name"
          className="w-full border p-3 rounded"
        />

        <input
          type="email"
          name="email"
          placeholder="Email Address"
          required
          className="w-full border p-3 rounded"
        />

        <input
          name="phone"
          placeholder="Phone Number"
          className="w-full border p-3 rounded"
        />

        <select
          name="service"
          required
          className="w-full border p-3 rounded"
        >
          <option value="">Select Service</option>
          <option>Web Development</option>
          <option>Mobile Development</option>
          <option>AI Solutions</option>
          <option>AWS Cloud</option>
          <option>Consulting</option>
        </select>

        <select
          name="budget"
          required
          className="w-full border p-3 rounded"
        >
          <option value="">Budget Range</option>
          <option>Under $5,000</option>
          <option>$5,000-$10,000</option>
          <option>$10,000-$25,000</option>
          <option>$25,000-$50,000</option>
          <option>$50,000+</option>
        </select>

        <textarea
          name="description"
          placeholder="Describe your project"
          required
          rows={6}
          className="w-full border p-3 rounded"
        />

        <input
          type="file"
          name="attachment"
          accept=".pdf,.doc,.docx,.png,.jpg"
          className="w-full border p-3 rounded"
        />

        <label className="flex gap-2">
          <input type="checkbox" required />
          <span>
            I agree to the Privacy Policy
          </span>
        </label>

        <button
          disabled={loading}
          className="bg-black text-white px-6 py-3 rounded"
        >
          {loading ? "Submitting..." : "Request Quote"}
        </button>
      </form>
    </div>
  );
}

Next.js → API Gateway → Lambda → DynamoDB → SES → CloudWatch + WAF + CAPTCHA

This is the architecture commonly used by agencies, SaaS companies, and enterprise consulting firms because it is scalable, secure, and inexpensive to operate.

Request for Quote (RFQ)

For a professional Request for Quote (RFQ) system using AWS + Node.js + Next.js, the goal is:

  1. Customer submits a quote request.
  2. System validates and blocks spam.
  3. Data is stored safely.
  4. Email notifications are sent.
  5. Admin can review all quote requests.
  6. System scales automatically and costs very little.

Validation (Zod)

Industry standard validation.


import { z } from "zod";

const QuoteSchema = z.object({
  name: z.string().min(2),
  email: z.email(),
  projectDescription: z.string().min(20)
});
