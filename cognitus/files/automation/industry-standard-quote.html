Production-grade AWS serverless “Request a Quote” system

Below is a production-grade AWS serverless “Request a Quote” system. This is the kind of architecture used in real SaaS backends.

You’ll get:

  • API Gateway (HTTP API)
  • Lambda (Node.js)
  • DynamoDB (storage + rate limit tracking)
  • SES (email delivery)
  • Cloudflare Turnstile CAPTCHA (bot protection)
  • CloudWatch + optional admin logging dashboard
  • IAM least-privilege setup
  • Clean scalable structure (ready for Serverless Framework or CDK)

🧠 1. Architecture Overview


Next.js frontend
      ↓

      ↓

📦 2. DynamoDB Tables

Table 1: QuoteRequests


PK: quoteId (string)
email: string
name: string
project: string
ip: string
createdAt: number
status: "NEW" | "SENT" | "FAILED"

Table 2: RateLimit


PK: rateKey (email+ip)
ttl: number (auto-expire after 24h)
count: number

⚙️ 3. Serverless Framework Setup

📁 serverless.yml


service: quote-system

provider:
  name: aws
  runtime: nodejs20.x
  region: ap-southeast-1
  environment:
    TABLE_QUOTES: QuoteRequests
    TABLE_RATE: RateLimit
    EMAIL_FROM: noreply@yourdomain.com
    EMAIL_TO: your@email.com

  iam:
    role:
      statements:
        - Effect: Allow
          Action:
            - dynamodb:PutItem
            - dynamodb:GetItem
            - dynamodb:UpdateItem
          Resource: "*"

        - Effect: Allow
          Action:
            - ses:SendEmail
          Resource: "*"

functions:
  submitQuote:
    handler: src/submitQuote.handler
    events:
      - httpApi:
          path: /quote
          method: POST

resources:
  Resources:

    QuoteTable:
      Type: AWS::DynamoDB::Table
      Properties:
        TableName: QuoteRequests
        BillingMode: PAY_PER_REQUEST
        AttributeDefinitions:
          - AttributeName: quoteId
            AttributeType: S
        KeySchema:
          - AttributeName: quoteId
            KeyType: HASH

    RateTable:
      Type: AWS::DynamoDB::Table
      Properties:
        TableName: RateLimit
        BillingMode: PAY_PER_REQUEST
        AttributeDefinitions:
          - AttributeName: rateKey
            AttributeType: S
        KeySchema:
          - AttributeName: rateKey
            KeyType: HASH

Backend verifies:


async function verifyCaptcha(token: string) {
    const res = await fetch(
        "https://challenges.cloudflare.com/turnstile/v0/siteverify",
        {
            method: "POST",
            headers: { "Content-Type": "application/x-www-form-urlencoded" },
            body: new URLSearchParams({
                secret: process.env.TURNSTILE_SECRET!,
                response: token,
            }),
        }
    );
    
    const data = await res.json();
    return data.success;
}

⚡ 5. Lambda Function (Core Logic)

📁 src/submitQuote.ts


import { DynamoDBClient } from "@aws-sdk/client-dynamodb";
import { DynamoDBDocumentClient, PutCommand, GetCommand } from "@aws-sdk/lib-dynamodb";
import { SESClient, SendEmailCommand } from "@aws-sdk/client-ses";
import { randomUUID } from "crypto";

const ddb = DynamoDBDocumentClient.from(new DynamoDBClient({}));
const ses = new SESClient({});

const TABLE_QUOTES = process.env.TABLE_QUOTES!;
const TABLE_RATE = process.env.TABLE_RATE!;

export const handler = async (event: any) => {
  try {
    const body = JSON.parse(event.body || "{}");

    const { name, email, project, captchaToken } = body;

    const ip =
      event.headers["x-forwarded-for"]?.split(",")[0] || "unknown";

    // 1. CAPTCHA VERIFY
    const captchaOk = await verifyCaptcha(captchaToken);
    if (!captchaOk) {
      return response(400, "Invalid CAPTCHA");
    }

    // 2. RATE LIMIT (1 per day per email+ip)
    const rateKey = `${email}:${ip}`;

    const rate = await ddb.send(
      new GetCommand({
        TableName: TABLE_RATE,
        Key: { rateKey },
      })
    );

    if (rate.Item) {
      return response(429, "Rate limit exceeded (1 per day)");
    }

    await ddb.send(
      new PutCommand({
        TableName: TABLE_RATE,
        Item: {
          rateKey,
          ttl: Math.floor(Date.now() / 1000) + 86400,
          count: 1,
        },
      })
    );

    // 3. STORE QUOTE
    const quoteId = randomUUID();

    await ddb.send(
      new PutCommand({
        TableName: TABLE_QUOTES,
        Item: {
          quoteId,
          name,
          email,
          project,
          ip,
          createdAt: Date.now(),
          status: "NEW",
        },
      })
    );

    // 4. SEND EMAIL (SES)
    await ses.send(
      new SendEmailCommand({
        Source: process.env.EMAIL_FROM!,
        Destination: { ToAddresses: [process.env.EMAIL_TO!] },
        Message: {
          Subject: { Data: `New Quote Request: ${name}` },
          Body: {
            Text: {
              Data: `
Name: ${name}
Email: ${email}

Project:
${project}

IP: ${ip}
              `,
            },
          },
        },
      })
    );

    return response(200, { success: true, quoteId });
  } catch (err: any) {
    console.error(err);
    return response(500, "Server error");
  }
};

function response(statusCode: number, body: any) {
  return {
    statusCode,
    body: JSON.stringify(body),
  };
}

async function verifyCaptcha(token: string) {
  const res = await fetch(
    "https://challenges.cloudflare.com/turnstile/v0/siteverify",
    {
      method: "POST",
      headers: { "Content-Type": "application/x-www-form-urlencoded" },
      body: new URLSearchParams({
        secret: process.env.TURNSTILE_SECRET!,
        response: token,
      }),
    }
  );

  const data = await res.json();
  return data.success;
}
