1. Node.js Backend

Install dependencies:


npm install express argon2 jsonwebtoken crypto dotenv

User Registration

Never store passwords directly.


const argon2 = require("argon2");

async function register(email, password) {
  const hashedPassword = await argon2.hash(password);

  await db.users.insertOne({
    email,
    password: hashedPassword
  });
}

Example stored record:


{
    "email": "user@example.com",
    "password": "$argon2id$v=19$m=65536,t=3,p=4$..."
}

Login Endpoint


const express = require("express");
const argon2 = require("argon2");

const app = express();

app.use(express.json());

app.post("/auth/login", async (req, res) => {
  const { email, password } = req.body;

  const user = await db.users.findOne({ email });

  if (!user) {
    return res.status(401).json({
      message: "Invalid credentials"
    });
  }

  const validPassword = await argon2.verify(
    user.password,
    password
  );

  if (!validPassword) {
    return res.status(401).json({
      message: "Invalid credentials"
    });
  }

  // Password valid
  // Trigger MFA

  res.json({
    mfaRequired: true
  });
});

3. Generate JWT Access Token


const jwt = require("jsonwebtoken");

function generateAccessToken(userId) {
  return jwt.sign(
    {
      sub: userId,
      type: "access"
    },
    process.env.JWT_SECRET,
    {
      expiresIn: "15m"
    }
  );
}

4. Generate Refresh Token

Use a cryptographically secure random token.


const crypto = require("crypto");

function generateRefreshToken() {
  return crypto.randomBytes(64).toString("hex");
}

Store only a hash of the refresh token:


const crypto = require("crypto");

function hashToken(token) {
  return crypto
    .createHash("sha256")
    .update(token)
    .digest("hex");
}

Database:


{
    "userId": "123",
    "refreshTokenHash": "abc123...",
    "expiresAt": "2026-06-30"
}

This protects users if your database is compromised.

5. Refresh Token Endpoint


app.post("/auth/refresh", async (req, res) => {
    const { refreshToken } = req.body;
    
    const tokenHash = hashToken(refreshToken);
    
    const record = await db.refreshTokens.findOne({
        tokenHash
    });
    
    if (!record) {
        return res.status(401).json({
        message: "Invalid token"
        });
    }
    
    const accessToken = generateAccessToken(
        record.userId
    );
    
    res.json({
        accessToken
    });
});

For higher security, rotate the refresh token every time it is used.

6. Protected Route

Middleware:


const jwt = require("jsonwebtoken");

function authenticate(req, res, next) {
  const authHeader = req.headers.authorization;

  if (!authHeader) {
    return res.sendStatus(401);
  }

  const token = authHeader.replace(
    "Bearer ",
    ""
  );

  try {
    req.user = jwt.verify(
      token,
      process.env.JWT_SECRET
    );

    next();
  } catch {
    return res.sendStatus(401);
  }
}

Usage:


app.get(
  "/accounts",
  authenticate,
  async (req, res) => {
    res.json({
      balance: 1000
    });
  }
);

npm install react-native-keychain

MFA Verification


async function verifyMfa(email, otp) {
    const response = await fetch(
        "https://api.example.com/auth/verify-mfa",
        {
        method: "POST",
        headers: {
            "Content-Type": "application/json"
        },
        body: JSON.stringify({
            email,
            otp
        })
        }
    );
    
    const data = await response.json();
    
    await Keychain.setGenericPassword(
        "token",
        JSON.stringify({
        accessToken: data.accessToken,
        refreshToken: data.refreshToken
        })
    );
}

Uses the iOS Keychain.

const credentials =

Never store tokens in:


AsyncStorage ❌
Redux Persist ❌
MMKV (unencrypted) ❌

For financial applications, use Keychain/Keystore-backed secure storage.

10. Auto Refresh Access Token

When the access token expires:


if (response.status === 401) {
    const newAccessToken =
        await refreshAccessToken();
    
    // Retry request
}

Refresh endpoint:


async function refreshAccessToken() {
    const credentials =
        await Keychain.getGenericPassword();
    
    const tokens = JSON.parse(
        credentials.password
    );
    
    const response = await fetch(
        "https://api.example.com/auth/refresh",
        {
        method: "POST",
        headers: {
            "Content-Type": "application/json"
        },
        body: JSON.stringify({
            refreshToken:
            tokens.refreshToken
        })
        }
    );
    
    const data = await response.json();
    
    tokens.accessToken =
        data.accessToken;
    
    await Keychain.setGenericPassword(
        "auth",
        JSON.stringify(tokens)
    );
    
    return data.accessToken;
}

Financial-Industry Hardening

For banking or fintech apps, add:

  • Argon2id password hashing.
  • MFA (TOTP via apps like Google Authenticator or SMS as a fallback).
  • TLS 1.2+ with certificate pinning.
  • Access tokens ≤ 15 minutes.
  • Refresh token rotation.
  • Store only hashed refresh tokens.
  • Device binding for refresh tokens.
  • Rate limiting on login attempts.
  • Account lockout after repeated failures.
  • Audit logging.
  • JWT signing with asymmetric keys (RS256/ES256) rather than shared secrets.
  • Keychain/Keystore secure storage.
  • Backend token revocation support.
  • Detect rooted/jailbroken devices when appropriate.

This design is very close to what many modern financial platforms use today: short-lived JWT access tokens, rotating refresh tokens, Argon2id password hashing, MFA, and secure device storage.
